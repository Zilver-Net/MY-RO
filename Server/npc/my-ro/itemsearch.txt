-	script	Item_Search	-,{

	search_loop:
		mes .NPCName$;
		mes "กรุณากรอก ^0000FFItem ID^000000 ที่ต้องการค้นหา:";
		mes "ใส่ 0 เพื่อยกเลิก";
		next;
		input .@item_id;
		next;
		
		if (.@item_id == 0) { end; }
		if (.@item_id < 0 || getitemname(.@item_id) == "" || getitemname(.@item_id) == "null") {
			mes .NPCName$;
			mes "^FF0000ผิดพลาด: ไม่พบ Item ID นี้ในฐานข้อมูลของเซิร์ฟเวอร์^000000";
			next;
			goto search_loop;
		}
		
		.@item_name$ = getitemname(.@item_id);
		.@item_type = getiteminfo(.@item_id, ITEMINFO_TYPE);
		.@can_buy = 1;
		.@reject_reason$ = "";
		
		if (.@item_name$ == "Unknown Item") {
			.@can_buy = 0;
			.@reject_reason$ = "ไอเทมนี้เป็น Unknown Item (ระบบปิดการซื้อเพื่อความปลอดภัย)";
		}
		
		if (.@can_buy == 1) {
			.@is_allowed = 0;
			
			if (.@item_type == 4 || .@item_type == 5 || .@item_type == 12) {
				.@is_allowed = 1;
			}
			else if (compare(.@item_name$, "Stone")) {
				.@is_allowed = 1;
			}
			else if (.@item_type == 6 && !compare(.@item_name$, "Card")) {
				.@is_allowed = 1;
			}
			
			if (.@is_allowed == 0) {
				.@can_buy = 0;
				.@reject_reason$ = "ระบบเปิดให้ซื้อเฉพาะ อุปกรณ์สวมใส่, Costume, Shadowgear และกลุ่มหินออฟชั่นเท่านั้น";
			}
		}

		mes .NPCName$;
		mes "--- ผลการค้นหาที่พบ ---";
		mes "ชื่อไอเทม: <ITEM>" + getitemname(.@item_id) + "<INFO>" + .@item_id + "</INFO></ITEM>";
		mes "Item ID: ^0000FF" + .@item_id + "^000000";
		mes "-----------------------";
		
		if (.@can_buy == 0) {
			mes "สถานะระบบ: [ ^FF0000ปิดการซื้อขายสำหรับไอเทมนี้^000000 ]";
			mes "เหตุผล: ^FF0000" + .@reject_reason$ + "^000000";
			next;
			goto search_loop;
		}
		
		.@currency_name$ = "";
		if (.currency_type == 0) .@currency_name$ = "Zeny";
		else if (.currency_type == 1) .@currency_name$ = "Cash Point";
		else if (.currency_type == 2) .@currency_name$ = getitemname(.currency_item_id);
		
		mes "สถานะระบบ: [ ^008800อนุญาตให้สั่งซื้อได้^000000 ]";
		mes "ราคาตั้งขาย: ^0000FF" + .price + "^000000 " + .@currency_name$ + " / ชิ้น";
		mes "-----------------------";
		mes "ต้องการสั่งซื้อไอเทมชิ้นนี้หรือไม่?";
		next;
		
		if (select("~ ซื้อ:~ ค้นหาชิ้นอื่นต่อ") == 2) {
			goto search_loop;
		}
		
		mes .NPCName$;
		mes "กรุณาใส่จำนวนที่ต้องการซื้อ (จำกัดครั้งละ 1 - 100 ชิ้น):";
		mes "ใส่ 0 เพื่อยกเลิก";
		next;
		input .@amount;
		next;
		
		if (.@amount == 0) { end; }
		if (.@amount < 0 || .@amount > 100) {
			mes .NPCName$;
			mes "^FF0000ผิดพลาด: จำนวนไม่ถูกต้อง (ใส่ได้เฉพาะ 1 - 100 ชิ้น)^000000";
			next;
			goto search_loop;
		}
		
		.@total_cost = .price * .@amount;
		
		if (.currency_type == 0) {
			if (zeny < .@total_cost) {
				mes .NPCName$;
				mes "^FF0000ผิดพลาด: เงิน Zeny ไม่พอ! ต้องใช้ทั้งหมด " + callfunc("F_InsertComma", .@total_cost) + " Zeny^000000";
				close;
			}
		} else if (.currency_type == 1) {
			if (#CASHPOINTS < .@total_cost) {
				mes .NPCName$;
				mes "^FF0000ผิดพลาด: Cash Point ไม่พอ! ต้องใช้ทั้งหมด " + .@total_cost + " Cash^000000";
				close;
			}
		} else if (.currency_type == 2) {
			if (countitem(.currency_item_id) < .@total_cost) {
				mes .NPCName$;
				mes "^FF0000ผิดพลาด: คุณมี " + .@currency_name$ + " ไม่เพียงพอ! ต้องใช้ " + .@total_cost + " ชิ้น^000000";
				close;
			}
		}
		
		if (!checkweight(.@item_id, .@amount)) {
			mes .NPCName$;
			mes "^FF0000ผิดพลาด: น้ำหนักในตัวเกิน หรือช่องเก็บของเต็มแล้ว!^000000";
			close;
		}
		
		if (.currency_type == 0) set zeny, zeny - .@total_cost;
		else if (.currency_type == 1) set #CASHPOINTS, #CASHPOINTS - .@total_cost;
		else if (.currency_type == 2) delitem .currency_item_id, .@total_cost;
		
		getitem .@item_id, .@amount;
		specialeffect2 EF_REPAIRWEAPON;
		
		mes .NPCName$;
		mes "การสั่งซื้อเสร็จสิ้นเรียบร้อย!";
		mes "ได้รับ ^0000FF" + .@item_name$ + "^000000 จำนวน " + .@amount + " ชิ้น";
		next;
		
		if (select("~ ค้นหาไอเทมชิ้นอื่นต่อ:~ ปิดหน้าต่างระบบ") == 1) {
			goto search_loop;
		}
		close;

	OnInit:
		.NPCName$ = "[ ^000080Item Search^000000 ]";
		.price = 20;              
		.currency_type = 1;       
		.currency_item_id = 674;  
		end;
}
