-	script	levelcheck	-,{	

	OnPCBaseLvUpEvent:
		// Check if player is below level 100 and apply EXP buff if enabled		
		callfunc "getexpbuff";
		// End of EXP buff system for new players
		if(DoIntro) { 
			end; 
		} else if(readparam(BaseLevel) >= 12){

			mes "ถึงเวลาออกผจญภัย !";

			} else
		end;
}

	end;
}
