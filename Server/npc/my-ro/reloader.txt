-	script	Reloader	-,{

	OnInit:
		bindatcmd "r", strnpcinfo(3)+"::OnAtcommandReload", 0;
		end;

	OnAtcommandReload:
		if (.@atcmd_parameters$[0] == "") {
			dispbottom "กรุณาใส่ชื่อ NPC! ตัวอย่าง: @r jobmaster";
			end;
		}
	
		.@target_file$ = "npc/my-ro/" + .@atcmd_parameters$[0] + ".txt";
		dispbottom "[Reloader] กำลังรีโหลด: " + .@target_file$;
	
		atcommand "@unloadnpcfile " + .@target_file$;
		sleep2 100;
		atcommand "@loadnpc " + .@target_file$;
	
		dispbottom "[Reloader] รีโหลดสคริปต์เสร็จสิ้นเรียบร้อยแล้ว!";
		end;
}
