//= เพิ่ม NPC สำหรับแลกการ์ดเป็นเงิน Zeny
//= ผู้เล่นสามารถนำการ์ดที่ไม่ต้องการมาแลกเป็น Zeny ได้
//= ราคาการ์ดจะถูกตั้งไว้ที่ 1000 Zeny ต่อการ์ด 1 ใบ
//= ข้อควรระวัง ** ระบบจะตรวจการ์ดในกระเป๋าของผู้เล่นทั้งหมด และลบการ์ดที่นำมาแลกออกจากตัวละคร ดังนั้นควรตรวจสอบให้แน่ใจว่ามีการ์ดที่ต้องการแลกอยู่ในกระเป๋าก่อนทำการแลก

morocc,147,107,6	script	Rily	436,{
	getinventorylist;
	if(@inventorylist_count > 0){
		freeloop(1);
		
		for (.@i = @inventorylist_count - 1; .@i >= 0; .@i--){
			if(@inventorylist_fav[.@i])
			continue;
			
			.@item_id = @inventorylist_id[.@i];
			
			if(getiteminfo(.@item_id,2) != 6 && !compare(getitemname(.@item_id),"Card"))
			continue;
			
			.@sell_amount = @inventorylist_amount[.@i];
			.@finalize = (.@sell_amount * .multiplier);			
			set zeny,zeny+=.@finalize;
			delitem .@item_id,.@sell_amount;
			dispbottom "ท่านได้แลกการ์ด "+(getitemname(.@item_id))+" จำนวน "+callfunc("F_InsertComma",.@sell_amount)+" ใบ เป็นเงิน "+callfunc("F_InsertComma",.@finalize)+" zeny.";
		}

		freeloop(0);
	}
	if (.@sell_amount == 0){
		dispbottom "กรุณานำการ์ดที่ต้องการแลกมาใส่ในกระเป๋าก่อนนะคะ";
		dispbottom "ราคาการ์ดจะถูกตั้งไว้ที่ "+callfunc("F_InsertComma",.multiplier)+" zeny ต่อการ์ด 1 ใบ";
		dispbottom "ข้อควรระวัง ** ระบบจะตรวจการ์ดในกระเป๋าของผู้เล่นทั้งหมด และลบการ์ดที่นำมาแลกออกจากตัวละคร ดังนั้นควรตรวจสอบให้แน่ใจว่าท่านได้เก็บการ์ดที่ไม่ต้องต้องการแลกไว้ในคลังแล้วนะคะ";
		end;
	}
	
	dispbottom "ดำเนินการเรียบร้อยค่ะ";
	end;
	
	OnInit:
		waitingroom ":: Card to Zeny :: ",0;
		
		.multiplier = 5000; // ราคาการ์ด
		
		end;
}
