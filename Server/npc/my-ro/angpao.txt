function	script	angpaw	{

	.@chance = rand(1, 10000);
	
	if (.@chance <= 10) {							// 0.10% chance to get big rewards (10M - 100M)
		.@get_zeny = rand(10000000, 100000000);
	} else if (.@chance <= 100) {					// 1.00% chance to get (1M - 10M)
		.@get_zeny = rand(1000000, 10000000);
	} else if (.@chance <= 500) {					// 5.00% chance to get (100K - 1M)
		.@get_zeny = rand(100000, 1000000);
	} else if (.@chance <= 1000) {					// 10.00% chance to get (10K - 100K)
		.@get_zeny = rand(10000, 100000);
	} else {										// 90.00% chance to get (1K - 10K)
		.@get_zeny = rand(1000, 10000);
	}

	Zeny += .@get_zeny;

	if (.@chance <= 200) { 
		.@get_angpao = rand(1, 4);
		getitem 33004, .@get_angpao;
		dispbottom "ท่านได้รับซองอั่งเปาเพิ่ม " + .@get_angpao + " ซอง";
	}
	
	if (.@get_zeny >= 1000000) {
		announce strcharinfo(0) + " ได้รับเงินจากซองอั่งเปา " + callfunc("F_InsertComma", .@get_zeny) + " zeny.", 0xFF99FF; 
	} else {
		dispbottom "ท่านได้รับเงินจากซองอั่งเปา " + callfunc("F_InsertComma", .@get_zeny) + " zeny.";
	}
	end;
}
