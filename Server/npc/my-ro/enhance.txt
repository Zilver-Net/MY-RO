	setarray .@equip_card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	set .@equip_refine, getequiprefinerycnt(.@part); //save refinery count
	
	if((countitem(.@en_material) < 1) || (zeny < .@en_price)) {
		mes .@npc_name$;
		mes "ขออภัยด้วย !";
		mes "ท่านมีเงินไม่พอ หรือไม่มี <ITEM>"+getitemname(.@en_material)+"<INFO>"+.@en_material+"</INFO></ITEM> สำหรับการถอนการเสริมพลัง";
		close;
	}
	mes .@npc_name$;
	mes "ต้องใช้ <ITEM>"+getitemname(.@en_material)+"<INFO>"+.@en_material+"</INFO></ITEM> ^3399FF1^000000 ea.";
	mes "และเงิน ^3399FF"+callfunc("F_InsertComma",.@en_price)+"^000000 zeny ในการถอนการเสริมพลัง";
	mes "โปรดยืนยัน";
	next;
	if (select("~ ยืนยัน:~ ยกเลิก") == 2 ) { end; }
	mes .@npc_name$;
	mes "โปรดเลือก Slot ที่ต้องการ";
	mes "กรุณาใส่เป็นตัวเลข 1 - 4";
	if (.@equip_card[0] == 0) { mes "Slot 1. ว่าง"; }
	else {	mes "Slot 1. "+getitemname(.@equip_card[0])+""; }
	if (.@equip_card[1] == 0) { mes "Slot 2. ว่าง"; }
	else {	mes "Slot 2. "+getitemname(.@equip_card[1])+""; }
	if (.@equip_card[2] == 0) { mes "Slot 3. ว่าง"; }
	else {	mes "Slot 3. "+getitemname(.@equip_card[2])+""; }
	if (.@equip_card[3] == 0) { mes "Slot 4. ว่าง"; }
	else {	mes "Slot 4. "+getitemname(.@equip_card[3])+""; }
	input .@selectslot;
	next;
	if (.@selectslot == 0 || .@selectslot > 4) { mes .@npc_name$; mes "ท่านใส่ Slot ไม่ถูกต้อง"; end; }
	if (.@selectslot == 1 && .@equip_card[0] != 0) { getitem2 .@equip_id , 1 , 1 , .@equip_refine, 0 , 0 , .@equip_card[1], .@equip_card[2], .@equip_card[3]; }
	else if (.@selectslot == 2 && .@equip_card[1] != 0) { getitem2 .@equip_id , 1 , 1 , .@equip_refine, 0 , .@equip_card[0], 0 , .@equip_card[2], .@equip_card[3]; }
	else if (.@selectslot == 3 && .@equip_card[2] != 0) { getitem2 .@equip_id , 1 , 1 , .@equip_refine, 0 , .@equip_card[0], .@equip_card[1], 0 , .@equip_card[3]; }
	else if (.@selectslot == 4 && .@equip_card[3] != 0) { getitem2 .@equip_id , 1 , 1 , .@equip_refine, 0 , .@equip_card[0], .@equip_card[1], .@equip_card[2], 0; }
	else { mes .@npc_name$; mes "Slot ที่ท่านเลือกนั้นว่างอยู่แล้ว"; end; }
	specialeffect2 EF_REPAIRWEAPON;
	delequip .@part;
	delitem .@en_material,1;
	set zeny, zeny-=.@en_price;
	mes .@npc_name$;
	mes "ดำเนินการเสร็จสิ้น";
	mes "แล้วพบกันใหม่";
	close;
	
	OnInit:
		waitingroom ":: Enchance ::",0;
		end;
}
