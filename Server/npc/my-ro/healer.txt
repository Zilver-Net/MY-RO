-	script	Healer_master	-1,{

	.@BF_Delay = 600000;		// 60000 = 1 min
	.@BF_level = 10;
	.@BF_boost = 100;
	.@Buffs1 = 1;			// Also buff players? (1: yes / 0: no) AB only.
	.@Buffs2 = 0;			// Also buff players? (1: yes / 0: no) Buff ETC.
	.@Buffs3 = 0;			// Also buff players? (1: yes / 0: no) Buff EXP/BRAGI.
	.@Indentified = 0;		// Also indentified item? (1: yes / 0: no)
	.@Delay = 3;			// Heal delay, in seconds
	.@Food = 0;				// Also food buff for players? (1: yes / 0: no)
	.@CP = 0;				// Also buff CP players? (1: yes / 0: no)

	if (@HD > gettimetick(2))
		end;

	percentheal 100,100;
	npcskill "AL_HEAL",11,99,60;

	sc_end SC_STONE;
	sc_end SC_FREEZE;
	sc_end SC_STUN;
	sc_end SC_SLEEP;
	sc_end SC_POISON;
	sc_end SC_CURSE;
	sc_end SC_SILENCE;
	sc_end SC_CONFUSION;
	sc_end SC_BLIND;
	sc_end SC_Hallucination;
	sc_end SC_BLEEDING;
	sc_end SC_DPOISON;
	sc_end SC_HELLPOWER;
	sc_end SC_ORCISH;
	sc_end SC_STAN;
	sc_end SA_REVERSEORCISH;

	if (.@Buffs1) {
		sc_start SC_BLESSING,.@BF_Delay,.@BF_level;
		sc_start SC_INCREASEAGI,.@BF_Delay,.@BF_level;
		sc_start4 SC_POEMBRAGI,.@BF_Delay,.@BF_level,80,100,0;
		sc_start4 SC_SERVICE4U,.@BF_Delay,.@BF_level,40,40,0;
		sc_start SC_EXPBOOST,.@BF_Delay,.@BF_boost+BaseLevel;
		sc_start SC_JEXPBOOST,.@BF_Delay,.@BF_boost+BaseLevel;
	}
	if (.@Buffs2) {
		sc_start SC_IMPOSITIO,.@BF_Delay,.@BF_level;
		sc_start SC_GLORIA,.@BF_Delay,.@BF_level;
		sc_start SC_ASSUMPTIO,.@BF_Delay,.@BF_level;
		sc_start SC_ANGELUS,.@BF_Delay,.@BF_level;
		sc_start SC_MAGNIFICAT,.@BF_Delay,.@BF_level;
	}
	if (.@Buffs3) {
		sc_start SC_EXPBOOST,.@BF_Delay,.@BF_boost+BaseLevel;
		sc_start SC_JEXPBOOST,.@BF_Delay,.@BF_boost+BaseLevel;
		sc_start SC_ITEMBOOST,.@BF_Delay,.@BF_boost;
		sc_start4 SC_POEMBRAGI,.@BF_Delay,.@BF_level,80,100,0;
		sc_start4 SC_SERVICE4U,.@BF_Delay,.@BF_level,40,40,0;
	}
	if (.@CP) {
		sc_start SC_CP_WEAPON,.@BF_Delay,.@BF_level;
		sc_start SC_CP_SHIELD,.@BF_Delay,.@BF_level;
		sc_start SC_CP_ARMOR,.@BF_Delay,.@BF_level;
		sc_start SC_CP_HELM,.@BF_Delay,.@BF_level;
	}
	if (.@Food) {
		sc_start SC_STRFood,.@BF_Delay,.@BF_level;
		sc_start SC_AGIFood,.@BF_Delay,.@BF_level;
		sc_start SC_VITFood,.@BF_Delay,.@BF_level;
		sc_start SC_INTFood,.@BF_Delay,.@BF_level;
		sc_start SC_DEXFood,.@BF_Delay,.@BF_level;
		sc_start SC_LUKFood,.@BF_Delay,.@BF_level;
		sc_start SC_HITFood,.@BF_Delay,.@BF_level;
		sc_start SC_FLEEFood,.@BF_Delay,.@BF_level;
		sc_start SC_BATKFood,.@BF_Delay,.@BF_level;
		sc_start SC_MATKFood,.@BF_Delay,.@BF_level;
		sc_start SC_COMMONSC_RESIST,.@BF_Delay,.@BF_level;
		//sc_start SC_ASPDPOTION2,.@BF_Delay,.@BF_level;
	}

	if (getbrokenid(1) > 0) { 
		repairall;
	}

	if (.@Indentified) {
		if (identifyall(false) > 0) {
			identifyall(true);
		}
	}
	
	if (.@Delay)
		@HD = gettimetick(2) + .@Delay;
	end;
}
