morocc,149,105,6	script	Lily	436,{
	.@coin = 33999; // Gold coin
	.@havecoin = countitem(.@coin);
	if (.@havecoin == 0) {
		dispbottom "คุณไม่มี "+getitemname(.@coin)+" ค่ะ";
		end;
	} else
		delitem .@coin,.@havecoin;
		set #CASHPOINTS,#CASHPOINTS+=.@havecoin;
		dispbottom "ดำเนินการเรียบร้อยค่ะ";
		dispbottom "คุณได้รับ "+.@havecoin+" Cash Points. | คงเหลือ "+callfunc("F_InsertComma",(#CASHPOINTS))+" Points.";
		end;
	
	OnInit:
		waitingroom ":: Gold Coin to Cash ::",0;
		end;
}

function	script	getmycash	{
	set #CASHPOINTS,#CASHPOINTS+=1;
	dispbottom "คุณได้รับ 1 Cash Points. | คงเหลือ "+callfunc("F_InsertComma",(#CASHPOINTS))+" Points.";
	end;
}
