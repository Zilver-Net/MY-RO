-	script	Coin_to_Cash	-,{

	OnTouchNPC:
		callfunc "getmycash", 1;
		end;
		
	OnInit:
		waitingroom ":: Gold Coin to Cash ::", 0;
		end;
}

function	script	getmycash	{
	
	.@coin = 33999; 					// Gold coin
	.@is_npc = getarg(0, 0); 			// from NPC = 1, from item = 0
	.@havecoin = countitem(.@coin);
	
	.@cash_to_add = .@havecoin + (.@is_npc ? 0 : 1);

	if (.@is_npc && .@havecoin == 0) {
		dispbottom "ท่านไม่มี " + getitemname(.@coin) + " ค่ะ";
		end;
	}

	if (.@cash_to_add > 2147483647 - #CASHPOINTS) {
		dispbottom "ขออภัยค่ะ! ไม่สามารถแลกเพิ่มได้เนื่องจาก Cash Points ของท่านจะเกินขีดจำกัดสูงสุด";
		
		if (!.@is_npc) { 
			getitem .@coin, 1; 
		}
		end;
	}

	if (.@havecoin > 0) {
		delitem .@coin, .@havecoin;
	}
	
	#CASHPOINTS += .@cash_to_add;
	dispbottom "ท่านได้รับ " + callfunc("F_InsertComma", .@cash_to_add) + " Cash Points. | คงเหลือ " + callfunc("F_InsertComma", #CASHPOINTS) + " Points.";
	end;
}
