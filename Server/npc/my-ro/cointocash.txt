morocc,149,105,6	script	Lily	436,{
	OnTouchNPC:
		@getmycashonnpc = 1;
		callfunc "getmycash";
		end;
	OnInit:
		waitingroom ":: Gold Coin to Cash ::",0;
		end;
}

function	script	getmycash	{
	.@coin = 33999; // Gold coin
	.@havecoin = countitem(.@coin);
	.@useditem = 1;
	if (.@havecoin == 0) {
		dispbottom "ท่านไม่มี "+getitemname(.@coin)+" ค่ะ";
		end;
	} else if (@getmycashonnpc) { 
		.@useditem = 0; 
	}
		delitem .@coin,.@havecoin;
		set #CASHPOINTS,#CASHPOINTS+=.@havecoin+=.@useditem;
		dispbottom "ท่านได้รับ "+callfunc("F_InsertComma",(.@havecoin))+" Cash Points. | คงเหลือ "+callfunc("F_InsertComma",(#CASHPOINTS))+" Points.";
		@getmycashonnpc = 0;
		end;
}
