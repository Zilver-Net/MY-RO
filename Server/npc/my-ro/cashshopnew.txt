// คอร์เอนจิ้นหลัก - ใช้โครงสร้างแยกสวิตช์เมนูมาตรฐาน ปลอดภัย ไร้ค้าง

function	script	F_CashShop_Engine	{
	set .@main_cat, getarg(0);
	set .@chosen_cat$, "";

	mes "[ Cash Shop ]";
	mes "กรุณาเลือกหมวดหมู่ที่ต้องการสั่งซื้อ:";
	
	// [STEP 1] แตกเมนูแบบ Hard-coded แยกตาม NPC เพื่อแก้ปัญหาเมนูขึ้นอันเดียว
	switch(.@main_cat) {
		case 1: // Equipment
			set .@s, select("~ Armor", "~ Shield", "~ Garment", "~ Footgear", "~ Others");
			if (.@s == 1) .@chosen_cat$ = "armor";
			if (.@s == 2) .@chosen_cat$ = "shield";
			if (.@s == 3) .@chosen_cat$ = "garment";
			if (.@s == 4) .@chosen_cat$ = "footgear";
			if (.@s == 5) .@chosen_cat$ = "others";
			break;
		case 2: // Accessory
			.@chosen_cat$ = "accessory";
			break;
		case 3: // ShadowGear
			.@chosen_cat$ = "shadowgear";
			break;
		case 4: // Headgear
			set .@s, select("~ Upper Headgear", "~ Middle Headgear", "~ Lower Headgear", "~ Headgear All");
			if (.@s == 1) .@chosen_cat$ = "upper_headgear";
			if (.@s == 2) .@chosen_cat$ = "middle_headgear";
			if (.@s == 3) .@chosen_cat$ = "lower_headgear";
			if (.@s == 4) .@chosen_cat$ = "headgear_all";
			break;
		case 5: // Costume
			set .@s, select("~ Costume Upper", "~ Costume Middle", "~ Costume Lower", "~ Costume Garment", "~ Costume Headgear All");
			if (.@s == 1) .@chosen_cat$ = "costume_upper";
			if (.@s == 2) .@chosen_cat$ = "costume_middle";
			if (.@s == 3) .@chosen_cat$ = "costume_lower";
			if (.@s == 4) .@chosen_cat$ = "costume_garment";
			if (.@s == 5) .@chosen_cat$ = "costume_headgear_all";
			break;
		case 6: // Weapon
			set .@s, select("~ Axe", "~ Book", "~ Bow", "~ Dagger/Knife", "~ Firearms", "~ Huuma", "~ Katar", "~ Knuckle", "~ Mace", "~ Musical/Whip", "~ Spear", "~ Staff", "~ Sword");
			if (.@s == 1) .@chosen_cat$ = "weapon_axe";
			if (.@s == 2) .@chosen_cat$ = "weapon_book";
			//============================== Fixed by GPT ==============================\\
			if (.@s == 3) .@chosen_cat$ = "weapon_bow";
			//============================== Old Code from Gemini 3.5 Extended==============================\\
			/*	
			if (.@s == 3) .@weapon_cat$ = "weapon_bow";
			*/
			if (.@s == 4) .@chosen_cat$ = "weapon_dagger_knife";
			if (.@s == 5) .@chosen_cat$ = "weapon_firearms";
			if (.@s == 6) .@chosen_cat$ = "weapon_huuma";
			if (.@s == 7) .@chosen_cat$ = "weapon_katar";
			if (.@s == 8) .@chosen_cat$ = "weapon_knuckle";
			if (.@s == 9) .@chosen_cat$ = "weapon_mace";
			if (.@s == 10) .@chosen_cat$ = "weapon_musical_whip";
			if (.@s == 11) .@chosen_cat$ = "weapon_spear";
			if (.@s == 12) .@chosen_cat$ = "weapon_staff";
			if (.@s == 13) .@chosen_cat$ = "weapon_sword";
			break;
	}

	next;

	//============================== Fixed by GPT ==============================\\
	// [STEP 2] เลือกตัวอักษรนำหน้าไอเทมแบบ Dynamic
		set .@alphabet_str$, getd("$@" + .@chosen_cat$ + "_chars$");

		if (.@alphabet_str$ == "") {
			debugmes "ERROR: No char data found for category [" + .@chosen_cat$ + "]";
			mes "[ Cash Shop ]";
			mes "Category data not found.";
			close;
		}

		set .@total_chars, explode(.@available_chars$, .@alphabet_str$, ":");

		if (.@total_chars <= 0) {
			debugmes "ERROR: explode() failed for [" + .@chosen_cat$ + "]";
			mes "[ Cash Shop ]";
			mes "Character list failed to load.";
			close;
		}

		mes "[ ค้นหาอักษรนำหน้า ]";

		set .@menu$, "";

		for (set .@i, 0; .@i < .@total_chars; set .@i, .@i + 1) {

			set .@c$, .@available_chars$[.@i];

			if (.@c$ == "0_9") {
				set .@display$, "~ 0-9";
			} else {
				set .@display$, "~ " + strtoupper(.@c$);
			}

			if (.@i > 0)
				set .@menu$, .@menu$ + ":";

			set .@menu$, .@menu$ + .@display$;
		}

		set .@char_choice, select(.@menu$) - 1;

		if (.@char_choice < 0 || .@char_choice >= .@total_chars) {
			close;
		}

		set .@chosen_char$, .@available_chars$[.@char_choice];
	//============================== Old Code from Gemini 3.5 Extended==============================\\
	/*
	// [STEP 2] เลือกตัวอักษรนำหน้าไอเทมแบบ Dynamic
	set .@alphabet_str$, getd("$@" + .@chosen_cat$ + "_chars$");
	setarray .@available_chars$[0], "";
	set .@total_chars, explode(.@available_chars$, .@alphabet_str$, ":");
	
	// ใช้โครงสร้างการแตกเมนูย่อยแบบวนลูปเข้าตัวแปรอาร์เรย์สคริปต์แทน String คั่นด้วยเครื่องหมายโคลอน
	mes "[ ค้นหาอักษรนำหน้า ]";
	setarray .@menu_display$[0], "";
	for (set .@i, 0; .@i < .@total_chars; set .@i, .@i + 1) {

		if (.@c$ == "0_9") {
			set .@menu_display$[.@i], "~ 0-9";
		} else {
			set .@pos, strpos("abcdefghijklmnopqrstuvwxyz", .@c$);
			set .@upper_c$, (.@pos != -1) ? charat("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@pos) : .@c$;
			set .@menu_display$[.@i], "~ " + .@upper_c$;
		}
	}
	
	// เรียก select แบบเลือกจากก้อนอาร์เรย์ตรงๆ (เอนจิ้น rAthena ยอมรับแบบนี้)
	set .@char_choice, select(.@menu_display$[0], .@menu_display$[1], .@menu_display$[2], .@menu_display$[3], .@menu_display$[4], .@menu_display$[5], .@menu_display$[6], .@menu_display$[7], .@menu_display$[8], .@menu_display$[9], .@menu_display$[10], .@menu_display$[11], .@menu_display$[12], .@menu_display$[13], .@menu_display$[14], .@menu_display$[15], .@menu_display$[16], .@menu_display$[17], .@menu_display$[18], .@menu_display$[19], .@menu_display$[20], .@menu_display$[21], .@menu_display$[22], .@menu_display$[23], .@menu_display$[24], .@menu_display$[25], .@menu_display$[26]) - 1;
	if (.@char_choice < 0) { close; end; }
	set .@chosen_char$, .@available_chars$[.@char_choice]; 
	*/

	// [STEP 3] คำนวณหน้าย่อย
	set .@max_pages, getd("$@" + .@chosen_cat$ + "_pages[" + .@char_choice + "]");
	set .@target_page, 1;
	
	if (.@max_pages > 1) {
		next;
		mes "[ เลือกหน้าย่อย ]";
		setarray .@page_display$[0], "";
		for (set .@p, 1; .@p <= .@max_pages; set .@p, .@p + 1) {
			set .@page_display$[.@p-1], "~ หน้า " + .@p;
		}
		set .@target_page, select(.@page_display$[0], .@page_display$[1], .@page_display$[2], .@page_display$[3], .@page_display$[4], .@page_display$[5], .@page_display$[6]);
		if (.@target_page <= 0) { close; end; }
	}

	// [STEP 4] จบประโยคสนทนาอย่างสมบูรณ์แบบด้วย close; แล้วปล่อยให้สคริปต์จบงาน ห้ามใช้ close2 ค้างไว้
	sprintf(.@shop_real_name$, "shop_%s_%s_%d", .@chosen_cat$, .@chosen_char$, .@target_page);
	
	// ใช้มาตรการความปลอดภัยสูงสุดในการกางร้านค้าของ rAthena เพื่อไม่ให้เกิดอาการ Task In Progress
	mes "กำลังเปิดร้านค้า...";	
	//============================== Fixed by GPT ==============================\\
	close2;
	//============================== Old Code from Gemini 3.5 Extended==============================\\
	/*
	close;
	*/
	// ให้เรียกเปิดร้านแยกหลังจากปิดหน้าต่างเรียบร้อยแล้ว
	callshop .@shop_real_name$, 1;
	end;
}

// คอร์เอนจิ้นหลัก - ใช้โครงสร้างแยกสวิตช์เมนูมาตรฐาน ปลอดภัย ไร้ค้าง
// ==========================================================
function	script	F_CashShop_Engine	{
	set .@main_cat, getarg(0);
	set .@chosen_cat$, "";

	mes "[ Cash Shop ]";
	mes "กรุณาเลือกหมวดหมู่ที่ต้องการสั่งซื้อ:";
	
	// [STEP 1] แตกเมนูแบบ Hard-coded แยกตาม NPC เพื่อแก้ปัญหาเมนูขึ้นอันเดียว
	switch(.@main_cat) {
		case 1: // Equipment
			set .@s, select("~ Armor", "~ Shield", "~ Garment", "~ Footgear", "~ Others");
			if (.@s == 1) .@chosen_cat$ = "armor";
			if (.@s == 2) .@chosen_cat$ = "shield";
			if (.@s == 3) .@chosen_cat$ = "garment";
			if (.@s == 4) .@chosen_cat$ = "footgear";
			if (.@s == 5) .@chosen_cat$ = "others";
			break;
		case 2: // Accessory
			.@chosen_cat$ = "accessory";
			break;
		case 3: // ShadowGear
			.@chosen_cat$ = "shadowgear";
			break;
		case 4: // Headgear
			set .@s, select("~ Upper Headgear", "~ Middle Headgear", "~ Lower Headgear", "~ Headgear All");
			if (.@s == 1) .@chosen_cat$ = "upper_headgear";
			if (.@s == 2) .@chosen_cat$ = "middle_headgear";
			if (.@s == 3) .@chosen_cat$ = "lower_headgear";
			if (.@s == 4) .@chosen_cat$ = "headgear_all";
			break;
		case 5: // Costume
			set .@s, select("~ Costume Upper", "~ Costume Middle", "~ Costume Lower", "~ Costume Garment", "~ Costume Headgear All");
			if (.@s == 1) .@chosen_cat$ = "costume_upper";
			if (.@s == 2) .@chosen_cat$ = "costume_middle";
			if (.@s == 3) .@chosen_cat$ = "costume_lower";
			if (.@s == 4) .@chosen_cat$ = "costume_garment";
			if (.@s == 5) .@chosen_cat$ = "costume_headgear_all";
			break;
		case 6: // Weapon
			set .@s, select("~ Axe", "~ Book", "~ Bow", "~ Dagger/Knife", "~ Firearms", "~ Huuma", "~ Katar", "~ Knuckle", "~ Mace", "~ Musical/Whip", "~ Spear", "~ Staff", "~ Sword");
			if (.@s == 1) .@chosen_cat$ = "weapon_axe";
			if (.@s == 2) .@chosen_cat$ = "weapon_book";
			if (.@s == 3) .@weapon_cat$ = "weapon_bow";
			if (.@s == 4) .@chosen_cat$ = "weapon_dagger_knife";
			if (.@s == 5) .@chosen_cat$ = "weapon_firearms";
			if (.@s == 6) .@chosen_cat$ = "weapon_huuma";
			if (.@s == 7) .@chosen_cat$ = "weapon_katar";
			if (.@s == 8) .@chosen_cat$ = "weapon_knuckle";
			if (.@s == 9) .@chosen_cat$ = "weapon_mace";
			if (.@s == 10) .@chosen_cat$ = "weapon_musical_whip";
			if (.@s == 11) .@chosen_cat$ = "weapon_spear";
			if (.@s == 12) .@chosen_cat$ = "weapon_staff";
			if (.@s == 13) .@chosen_cat$ = "weapon_sword";
			break;
	}

	next;

	// [STEP 2] เลือกตัวอักษรนำหน้าไอเทมแบบ Dynamic
	set .@alphabet_str$, getd("$@" + .@chosen_cat$ + "_chars$");
	setarray .@available_chars$[0], "";
	set .@total_chars, explode(.@available_chars$, .@alphabet_str$, ":");
	
	// ใช้โครงสร้างการแตกเมนูย่อยแบบวนลูปเข้าตัวแปรอาร์เรย์สคริปต์แทน String คั่นด้วยเครื่องหมายโคลอน
	mes "[ ค้นหาอักษรนำหน้า ]";
	setarray .@menu_display$[0], "";
	for (set .@i, 0; .@i < .@total_chars; set .@i, .@i + 1) {
		set .@c$, .@available_chars$[.@i];
		if (.@c$ == "0_9") {
			set .@menu_display$[.@i], "~ 0-9";
		} else {
			set .@pos, strpos("abcdefghijklmnopqrstuvwxyz", .@c$);
			set .@upper_c$, (.@pos != -1) ? charat("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@pos) : .@c$;
			set .@menu_display$[.@i], "~ " + .@upper_c$;
		}
	}
	
	// เรียก select แบบเลือกจากก้อนอาร์เรย์ตรงๆ (เอนจิ้น rAthena ยอมรับแบบนี้)
	set .@char_choice, select(.@menu_display$[0], .@menu_display$[1], .@menu_display$[2], .@menu_display$[3], .@menu_display$[4], .@menu_display$[5], .@menu_display$[6], .@menu_display$[7], .@menu_display$[8], .@menu_display$[9], .@menu_display$[10], .@menu_display$[11], .@menu_display$[12], .@menu_display$[13], .@menu_display$[14], .@menu_display$[15], .@menu_display$[16], .@menu_display$[17], .@menu_display$[18], .@menu_display$[19], .@menu_display$[20], .@menu_display$[21], .@menu_display$[22], .@menu_display$[23], .@menu_display$[24], .@menu_display$[25], .@menu_display$[26]) - 1;
	if (.@char_choice < 0) { close; end; }
	set .@chosen_char$, .@available_chars$[.@char_choice]; 

	// [STEP 3] คำนวณหน้าย่อย
	set .@max_pages, getd("$@" + .@chosen_cat$ + "_pages[" + .@char_choice + "]");
	set .@target_page, 1;
	
	if (.@max_pages > 1) {
		next;
		mes "[ เลือกหน้าย่อย ]";
		setarray .@page_display$[0], "";
		for (set .@p, 1; .@p <= .@max_pages; set .@p, .@p + 1) {
			set .@page_display$[.@p-1], "~ หน้า " + .@p;
		}
		set .@target_page, select(.@page_display$[0], .@page_display$[1], .@page_display$[2], .@page_display$[3], .@page_display$[4], .@page_display$[5], .@page_display$[6]);
		if (.@target_page <= 0) { close; end; }
	}

	// [STEP 4] จบประโยคสนทนาอย่างสมบูรณ์แบบด้วย close; แล้วปล่อยให้สคริปต์จบงาน ห้ามใช้ close2 ค้างไว้
	sprintf(.@shop_real_name$, "shop_%s_%s_%d", .@chosen_cat$, .@chosen_char$, .@target_page);
	
	// ใช้มาตรการความปลอดภัยสูงสุดในการกางร้านค้าของ rAthena เพื่อไม่ให้เกิดอาการ Task In Progress
	mes "กำลังเปิดร้านค้า...";
	close;
	
	// ให้เรียกเปิดร้านแยกหลังจากปิดหน้าต่างเรียบร้อยแล้ว
	callshop .@shop_real_name$, 1;
	end;
}
