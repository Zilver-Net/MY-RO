-	script	Kafra_master	-,{

	cutin "kafra_09",2;
	switch(select("~ Save:~ เปิดคลัง:~ เช่านก / รถเข็น ฯ:~ ถอดการ์ด:~ รีเซท:~ เพิ่มสกิลพิเศษ:~ แปลงเพศ:~ เปลื่ยนชื่อ:~ ตรวจสอบจำนวนผู้เล่น:~ Reset Quest Instant")) {
		case 1:
			getmapxy(@mapname$,.@x,.@y,BL_PC);
			dispbottom "บันทึกจุดเซฟเรียบร้อยค่ะ";
			save @mapname$,.@x,.@y,2,2;
			cutin "",255;
			end;

		case 2:
			openstorage;
			cutin "",255;
			end;

		case 3:
			if (ismounting()) {
				dispbottom "โปรดถอดพาหนะออกก่อนค่ะ";
				cutin "",255;
				end;
			} else if (((eaclass()&EAJ_THIRDMASK)==EAJ_RANGER) && !countitem(6124)) {
				if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) {
					if(select(" ~ Falcon: ~ Warg")==1) setfalcon;
						else getitem 6124,1; //Wolf's_Flute
					} else getitem 6124,1; //Wolf's_Flute
			} else if (((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) && !checkcart() && getskilllv("MC_PUSHCART")) {
				if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) {
					if(select(" ~ Cart: ~ Mado")==1) setcart;
						else setmadogear;
				} else setcart;
			} else if (!checkcart() && getskilllv("MC_PUSHCART")) setcart;
			else if (!checkfalcon() && getskilllv("HT_FALCON") && !checkwug()) setfalcon;
			else if (!checkriding() && !checkdragon() && getskilllv("KN_RIDING")) {
				if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) setdragon;
					else setriding;
			} else if (!checkmadogear() && getskilllv("NC_MADOLICENCE")) setmadogear;
				else {
					dispbottom "ท่านไม่มีสกิลค่ะ";
					cutin "",255;
					end;
				}
			specialeffect2 EF_TEIHIT3;
			cutin "",255;
			end;

		case 4:
			disable_items;
			dispbottom "โปรดเลือกอุปกรณ์ที่ต้องการค่ะ";
			setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
			for(.@i = 1; .@i<=10; ++.@i) {
				if (getequipisequiped(.@indices[.@i])) {
					.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
					.@equipped = 1;
				}
				.@menu$ = .@menu$ + ":";
			}
			.@part = .@indices[ select(.@menu$) ];
				switch(select("- ยืนยัน:- ยกเลิก")) {
					case 1:
						successremovecards .@part;
						dispbottom "ดำเนินการเรียบร้อยค่ะ";
						cutin "",255;
						end;
					case 2:
						cutin "",255;
						end;
				}

		case 5:
			//set .@prizeresetstat,BaseLevel*1000;
			//set .@prizeresetsskill,JobLevel*1000;
			dispbottom "กรุณาเลือกการรีเซทที่ต้องการค่ะ";
			switch(select("~ Status & Traits Status:~ Skill")) {
				case 1:
					/*if (zeny < .@prizeresetstat) {
						dispbottom "ท่านมีเงินไม่พอ";
						cutin "",255;
						end;
					} else
						set zeny,zeny-=.@prizeresetstat;*/
						resetstatus;
						if(IsReBirth) { set statuspoint,statuspoint+1173; }
						dispbottom "ดำเนินการเรียบร้อยค่ะ";
						cutin "",255;
						end;
				case 2:
					/*if (zeny < .@prizeresetsskill) {
						dispbottom "ท่านมีเงินไม่พอ";
						cutin "",255;
						end;
					} else
						set zeny,zeny-=.@prizeresetsskill;*/
						resetskill;
						if(IsReBirth) { set skillpoint,skillpoint+107; }					
						dispbottom "ดำเนินการเรียบร้อยค่ะ";
						cutin "",255;
						end;
			}

		case 6:
            AddSkillMenu:
                menu    "~ เพิ่มความจุน้ำหนัก ("+AddSkillWeightMemory+"/"+.AddSkillWeight[1]+")",AddSkillWeight,
                        "~ Greed ("+AddSkillGreedMemory+"/"+.AddSkillGreed[1]+")",AddSkillGreed,
                        "~ Heal ("+AddSkillHealMemory+"/"+.AddSkillHeal[1]+")",AddSkillHeal,
                        "~ Teleport ("+AddSkillTeleportMemory+"/"+.AddSkillTeleport[1]+")",AddSkillTeleport;

                AddSkillWeight:
                    if (AddSkillWeightMemory < .AddSkillWeight[1]) {
                        if (countitem(.AddSkillWeight[0]) >= .AddSkillWeight[2]) {
                            specialeffect2 EF_EARTHSPIKE; sleep2 300;
                            specialeffect2 EF_DEVIL; sleep2 300;
                            specialeffect2 EF_COIN; sleep2 300;
                            specialeffect2 EF_SIGHTRASHER;
                            dispbottom "ดำเนินการเรียบร้อยค่ะ";
                            delitem .AddSkillWeight[0],.AddSkillWeight[2];
                            set AddSkillWeightMemory, AddSkillWeightMemory + 1;
                            skill .AddSkillWeight[3],AddSkillWeightMemory,SKILL_PERM_GRANT;
                            
                            if (select("~ ดำเนินการต่อ") == 1) { 
                                goto AddSkillMenu; 
                            }
                            cutin "",255;
                            end;
                        } else {
                            dispbottom "ท่านไม่มี "+getitemname(.AddSkillWeight[0])+" ค่ะ";
                            cutin "",255;
                            end;
                        } 
                    } else {
                        dispbottom "ท่านถึงขีดจำกัดแล้วค่ะ";
                        cutin "",255;
                        end;
                    }

                AddSkillGreed:
                    if (AddSkillGreedMemory < .AddSkillGreed[1]) {
                        if (countitem(.AddSkillGreed[0]) >= .AddSkillGreed[2]) {
                            specialeffect2 EF_EARTHSPIKE; sleep2 300;
                            specialeffect2 EF_DEVIL; sleep2 300;
                            specialeffect2 EF_COIN; sleep2 300;
                            specialeffect2 EF_SIGHTRASHER;
                            dispbottom "ดำเนินการเรียบร้อยค่ะ";
                            delitem .AddSkillGreed[0],.AddSkillGreed[2];
                            set AddSkillGreedMemory, AddSkillGreedMemory + 1;
                            skill .AddSkillGreed[3],AddSkillGreedMemory,SKILL_PERM_GRANT;
                            
                            if (select("~ ดำเนินการต่อ") == 1) { 
                                goto AddSkillMenu; 
                            }
                            cutin "",255;
                            end;
                        } else {
                            dispbottom "ท่านไม่มี "+getitemname(.AddSkillGreed[0])+" ค่ะ";
                            cutin "",255;
                            end;
                        } 
                    } else {
                        dispbottom "ท่านถึงขีดจำกัดแล้วค่ะ";
                        cutin "",255;
                        end;
                    }

                AddSkillHeal:
                    if (AddSkillHealMemory < .AddSkillHeal[1]) {
                        if (countitem(.AddSkillHeal[0]) >= .AddSkillHeal[2]) {
                            specialeffect2 EF_EARTHSPIKE; sleep2 300;
                            specialeffect2 EF_DEVIL; sleep2 300;
                            specialeffect2 EF_COIN; sleep2 300;
                            specialeffect2 EF_SIGHTRASHER;
                            dispbottom "ดำเนินการเรียบร้อยค่ะ";
                            delitem .AddSkillHeal[0],.AddSkillHeal[2];
                            set AddSkillHealMemory, AddSkillHealMemory + 1;
                            skill .AddSkillHeal[3],AddSkillHealMemory,SKILL_PERM_GRANT;
                            
                            if (select("~ ดำเนินการต่อ") == 1) { 
                                goto AddSkillMenu; 
                            }
                            cutin "",255;
                            end;
                        } else {
                            dispbottom "ท่านไม่มี "+getitemname(.AddSkillHeal[0])+" ค่ะ";
                            cutin "",255;
                            end;
                        } 
                    } else {
                        dispbottom "ท่านถึงขีดจำกัดแล้วค่ะ";
                        cutin "",255;
                        end;
                    }

                AddSkillTeleport:
                    if (AddSkillTeleportMemory < .AddSkillTeleport[1]) {
                        if (countitem(.AddSkillTeleport[0]) >= .AddSkillTeleport[2]) {
                            specialeffect2 EF_EARTHSPIKE; sleep2 300;
                            specialeffect2 EF_DEVIL; sleep2 300;
                            specialeffect2 EF_COIN; sleep2 300;
                            specialeffect2 EF_SIGHTRASHER;
                            dispbottom "ดำเนินการเรียบร้อยค่ะ";
                            delitem .AddSkillTeleport[0],.AddSkillTeleport[2];
                            set AddSkillTeleportMemory, AddSkillTeleportMemory + 1;
                            skill .AddSkillTeleport[3],AddSkillTeleportMemory,SKILL_PERM_GRANT;
                            
                            if (select("~ ดำเนินการต่อ") == 1) { 
                                goto AddSkillMenu; 
                            }
                            cutin "",255;
                            end;
                        } else {
                            dispbottom "ท่านไม่มี "+getitemname(.AddSkillTeleport[0])+" ค่ะ";
                            cutin "",255;
                            end;
                        } 
                    } else {
                        dispbottom "ท่านถึงขีดจำกัดแล้วค่ะ";
                        cutin "",255;
                        end;
                    }

		case 7:
			switch(select("~ ชาย:~ หญิง")){
				case 1:
					query_sql("UPDATE `char` SET `sex` = 'M' WHERE `char`.`char_id` = '"+getcharid(0)+"'");
					dispbottom "ดำเนินการเรียบร้อยค่ะ";
					atcommand strcharinfo(0);
					cutin "",255;
					end;
				case 2:
					query_sql("UPDATE `char` SET `sex` = 'F' WHERE `char`.`char_id` = '"+getcharid(0)+"'");
					dispbottom "ดำเนินการเรียบร้อยค่ะ";
					atcommand strcharinfo(0);
					cutin "",255;
					end;
			}

		case 8:
			dispbottom "กรุณาใส่ชื่อที่ต้องการค่ะ";
			input .@namech$;
			if(.@namech$ == " ") goto failed;
			query_sql("UPDATE `char` SET `name` = '"+.@namech$+"' WHERE `char`.`char_id` = '"+getcharid(0)+"'");
			dispbottom "ดำเนินการเรียบร้อยค่ะ";
			cutin "",255;
			end;
			failed:
				dispbottom "ท่านใส่ชื่อไม่ถูกต้องค่ะ";
			cutin "",255;
			end;
		case 9:
			dispbottom "จำนวนผู้เล่นปัจจุบัน "+getusers(1)+" คน";
			cutin "",255;
			end;
		case 10:
			switch(select("~ Devil Tower:~ 100 Tower")){
				case 1:
					if (isbegin_quest(7568)) erasequest 7568;// Explore the tower
					if (isbegin_quest(7569)) erasequest 7569;// Treat the injured
					if (isbegin_quest(7570)) erasequest 7570;// Destroy the demons
					if (isbegin_quest(7571)) erasequest 7571;// Tower Expedition
					if (isbegin_quest(7572)) erasequest 7572;// Lucile...?
					if (isbegin_quest(7573)) erasequest 7573;// Magic Swordman Thanatos
					if (isbegin_quest(7574)) erasequest 7574;// Thanatos Tower
					if (isbegin_quest(7576)) erasequest 7576;// Morocc castle seal
					if (isbegin_quest(7577)) erasequest 7577;// Space distortion
					dispbottom "ทำการรีเซทดันเจี้ยน Devil Tower เรียบร้อยค่ะ";
					cutin "",255;
					end;
				case 2:
					if (isbegin_quest(60200) == 0) { 
						dispbottom "ท่านยังไม่ได้เริ่มดันเจี้ยน 100 ชั้นค่ะ";
						cutin "",255;
						end;
					} else
						set etower_timer,0;
						erasequest 60200;
						erasequest 60201;
						set party_instance_id,0;
						dispbottom "ทำการรีเซทดันเจี้ยน 100 ชั้นเรียบร้อยค่ะ";
						cutin "",255;
						end;
			}
	}
	
	OnInit:
        setarray .AddSkillWeight[0],7776,10,1,681;              //  Gym Pass,Maxlvl 10, Use per time 1, ALL_INCCARRY
        setarray .AddSkillGreed[0],33009,1,10,1013;             //  Skill Scroll, Maxlvl 1, Use per time 10, BS_GREED
        setarray .AddSkillHeal[0],33009,1,10,28;                //  Skill Scroll, Maxlvl 1, Use per time 10, AL_HEAL
        setarray .AddSkillTeleport[0],33009,1,10,26;            //  Skill Scroll, Maxlvl 1, Use per time 10, AL_TELEPORT
        end;
}
